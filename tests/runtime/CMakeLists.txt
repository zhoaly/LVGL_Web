cmake_minimum_required(VERSION 3.20)
project(ui_runtime_tests C CXX)
set(CMAKE_C_STANDARD 11)
enable_testing()
set(LV_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
if(NOT LVGL_SOURCE_DIR)
    message(FATAL_ERROR "Pass -DLVGL_SOURCE_DIR=<LVGL 9.3 source>")
endif()
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
include("${CMAKE_CURRENT_LIST_DIR}/../../src/lvgl_app/lvgl_app_sources.cmake")
add_executable(ui_runtime test_runtime.c ${LVGL_APP_CORE_SOURCES}
    ../../src/simulator/App_UiWifiMock.c)
target_include_directories(ui_runtime PRIVATE stubs ../../src ${LVGL_APP_INCLUDE_DIRS})
target_link_libraries(ui_runtime PRIVATE lvgl)
add_test(NAME wifi_ui_runtime COMMAND ui_runtime)
